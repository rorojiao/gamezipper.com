def build_grid(n, filled):
    """filled is a list of (r,c) tuples for the upper-left half.
    Mirroring: for each (r,c), also fill (n-1-r, n-1-c).
    For odd n, the center (m,m) is its own partner.
    For even n, the center quartet (m,m), (m,m-1), (m-1,m), (m-1,m-1)
    forms a single symmetric block where the 4 cells must all match
    (so we mirror within the half)."""
    g = [[0] * n for _ in range(n)]
    mid = (n - 1) / 2.0
    for r, c in filled:
        g[r][c] = 1
        # symmetric partner
        rr, cc = n - 1 - r, n - 1 - c
        g[rr][cc] = 1
    return g
